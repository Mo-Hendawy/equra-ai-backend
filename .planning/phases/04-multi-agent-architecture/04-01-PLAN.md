**CREATE server/agents/orchestrator.ts** — sequences pipeline with 30s per-agent timeouts (ARCH-06).

```typescript
const AGENT_TIMEOUT_MS = 30_000;

function withTimeout<T>(promise: Promise<T>, ms: number, agentName: string): Promise<T | null> {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), ms);
  return promise
    .finally(() => clearTimeout(timer))
    .catch(err => {
      console.warn(`[orchestrator] ${agentName} timed out or failed — continuing without output`);
      return null;
    });
}

export class AgentOrchestrator {
  async run(input: DataAgentInput): Promise<PipelineResult> {
    // Step 1: DataAgent (no timeout — deterministic fetches, no LLM)
    console.log(`[orchestrator] Pipeline start — ${input.symbol}`);
    const dataOutput = await dataAgent.run(input);

    // Step 2: AnalysisAgent — 30s timeout
    const analysisOutput = await withTimeout(
      analysisAgent.run({ stockDataForAI: dataOutput.stockDataForAI, episodicContext: dataOutput.episodicContext, refresh: input.refresh }),
      AGENT_TIMEOUT_MS, 'AnalysisAgent'
    );

    if (!analysisOutput) {
      // AnalysisAgent failed — return null analysis, pipeline cannot continue to Decision
      // Return a PipelineResult with null analysis so the caller can fall back
      return { dataOutput, analysisOutput: null, decisionOutput: { finalRecommendation: 'Hold', adjustedConfidence: 'Low', criticFeedback: null, decisionId: null } };
    }

    // Step 3: CriticAgent — 30s timeout, fail-open (C4 pitfall prevention)
    const criticOutput = await withTimeout(
      criticAgent.run({ analysis: analysisOutput, stockData: dataOutput.stockDataForAI }),
      AGENT_TIMEOUT_MS, 'CriticAgent'
    );
    // criticOutput may be null (fail-open) — DecisionAgent handles null fine

    // Step 4: DecisionAgent — 30s timeout
    const decisionOutput = await withTimeout(
      decisionAgent.run({
        symbol: input.symbol,
        analysis: analysisOutput,
        criticFeedback: criticOutput,
        stockDataForAI: dataOutput.stockDataForAI,
        computedFields: dataOutput.computedFields,
        priceSource: input.price.source,
        financialsSource: input.financials.source,
      }),
      AGENT_TIMEOUT_MS, 'DecisionAgent'
    );

    console.log(`[orchestrator] Pipeline complete — ${input.symbol}`);
    return {
      dataOutput,
      analysisOutput,
      decisionOutput: decisionOutput ?? { finalRecommendation: analysisOutput.recommendation, adjustedConfidence: analysisOutput.confidence, criticFeedback: criticOutput, decisionId: null },
    };
  }
}
export const orchestrator = new AgentOrchestrator();
```
