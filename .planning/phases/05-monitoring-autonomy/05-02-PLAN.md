    ```typescript
    // AUTO-02: Reject stale price data (>2 hours old)
    if (input.price.priceTimestamp) {
      const ageMs = Date.now() - new Date(input.price.priceTimestamp).getTime();
      const twoHoursMs = 2 * 60 * 60 * 1000;
      if (ageMs > twoHoursMs) {
        log.warn({ symbol: input.symbol, ageMs, priceTimestamp: input.price.priceTimestamp }, 'AUTO-02: stale price rejected — data >2 hours old');
        return {
          dataOutput: { stockDataForAI: {} as any, episodicContext: undefined, computedFields: { sharpeRatio: null, sortinoRatio: null, eps: null, peRatio: null, bookValue: null, priceToBook: null, dividendYield: null } },

          stalePriceRejected: true,
          flaggedForReview: false,
        };
      }
    }
    ```
